/**
 * Escape XML special characters, preserving existing HTML/XML entities.
 */
function escapeXml(text: string): string {
  return text
    .replace(/&(?!(?:#\d+|#x[0-9a-fA-F]+|[a-zA-Z]+);)/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&apos;');
}
 
/**
 * Map file-suffix language codes to proper BCP-47 hreflang codes.
 * Norwegian files use the suffix "no" but hreflang should be "nb" (Bokmål).
 */
function hreflangCode(lang: string): string {
  if (lang === 'no') return 'nb';
  return lang;
}
 
/**
 * Derive a stable publication date from the filename date prefix (YYYY-MM-DD)
 * or fall back to the file's modification time. Never uses "now" so builds are
 * deterministic.
 */
function stablePubDate(filePath: string): string {
  const basename = path.basename(filePath);
  const dateMatch = basename.match(/^(\d{4}-\d{2}-\d{2})/);
  if (dateMatch) {
    return new Date(`${dateMatch[1]}T12:00:00Z`).toISOString();
  }
  try {
    return fs.statSync(filePath).mtime.toISOString();
  } catch (_e: unknown) {
    return '2026-01-01T12:00:00.000Z';
  }
}
